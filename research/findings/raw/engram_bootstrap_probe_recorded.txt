from __future__ import annotations

import numpy as np
from sim.kernels import fused_eligibility_trace_decay  # REUSED UNMODIFIED

# Schedule constants (NOT science bars). N>=12 so chance is low and
# the absolute control bar is unambiguous (the small-N artifact
# structurally excluded, per the prior validated probes).
_N = 12             # concept pools; chance = 1/12 ~ 0.083
_P = 64             # units per sparse population code
_K = 8              # active units per code
_GAP = 6            # temporal gap (decision -> delayed reward)
_GAMMA = 0.95
_LAMBDA = 0.9
_ALPHA = 0.15
_EPS = 0.10
_NOISE = 0.10       # readout noise (the v16/in-bridge harder regime)
_BIND = 3.0         # engram one-shot co-activation bind strength
_EPOCHS = 60
_N_TR = 1500

# Pre-registered FROZEN bars (NEVER tuned). chance = 1/_N ~ 0.083.
_V1_REWARD_FRAC_MIN = 0.20   # bootstrap dissolves n_rewarded=0 (>> cold ~0)
_SCI_ACC_MIN = 0.80          # engram_plus_td genuinely generative
_SCI_GAIN_MIN = 0.25         # generative gain over retrieval-only engram
_CTRL_ACC_MAX = 0.40         # cold/permuted/wrongsign ~chance+margin
_NOTRACE_GAIN_MAX = 0.10     # no_trace must NOT add the generative gain
_MIN_SEEDS = 3


def _codes(rng):
    X = np.zeros((_N, _P))
    for c in range(_N):
        X[c, rng.choice(_P, _K, replace=False)] = 1.0
    return X


def _engram_bind(W, codes, pi):
    """Validated one-shot engram bind (Tonegawa D.14 abstract analog):
    reward-FREE Hebbian co-activation of the (cue_i, target pi[i])
    ensemble -- adds a fixed prior to W[cue_units, pi[i]]. NO reward
    used (this is exactly why it dissolves the n_rewarded=0 cold-start
    blocker)."""
    for i in range(_N):
        W[:, pi[i]] += _BIND * codes[i] / _K
    return W


def _run(mode, seed):
    """Returns (gap_acc, reward_frac). Episode: present cue_i sparse
    code; eps-greedy decision over targets from the noisy readout;
    TEMPORAL GAP; reward at t_R iff selected==pi[i]; TD(lambda)+
    eligibility refinement via the bridge-validated decay. `cold` =
    no engram bootstrap (the in-bridge-VOID analog). `engram_only` =
    bind, no TD. `no_trace` = bind + TD but eligibility zeroed across
    the gap. `permuted_boot` = engram binds a WRONG mapping."""
    rng = np.random.default_rng(seed)
    pi = rng.permutation(_N)
    codes = _codes(rng)
    W = np.zeros((_P, _N))
    if mode in ("engram_only", "engram_plus_td", "no_trace",
                "wrongsign"):
        _engram_bind(W, codes, pi)
    elif mode == "permuted_boot":
        _engram_bind(W, codes, rng.permutation(_N))   # WRONG mapping
    do_td = mode in ("engram_plus_td", "no_trace", "permuted_boot",
                     "wrongsign", "cold")
    decay = _GAMMA * _LAMBDA
    rew = 0
    tot = 0
    if do_td:
        for _t in range(_N_TR):
            i = int(rng.integers(_N))
            x = codes[i]
            logits = x @ W + rng.normal(0, _NOISE, _N)
            b = (int(rng.integers(_N)) if rng.random() < _EPS
                 else int(np.argmax(logits)))
            e = np.zeros((_P, _N))
            e[:, b] = x
            q = x @ W[:, b]
            for _g in range(_GAP):
                if mode == "no_trace":
                    e[:] = 0.0
                else:
                    e = np.asarray(
                        fused_eligibility_trace_decay(e, decay))
            r = 1.0 if b == pi[i] else 0.0
            rew += int(r > 0.5)
            tot += 1
            d = r + _GAMMA * 0.0 - q
            step = -d if mode == "wrongsign" else d
            W = W + _ALPHA * step * e
    # noise-free greedy gapped eval over all cues
    acc = float(np.mean(np.argmax(codes @ W, axis=1) == pi))
    rfrac = (rew / tot) if tot else 0.0
    return acc, rfrac

              "permuted_boot=%.3f wrongsign=%.3f"

        print("  >>> VERDICT: VOID (<%d seeds)." % _MIN_SEEDS)
